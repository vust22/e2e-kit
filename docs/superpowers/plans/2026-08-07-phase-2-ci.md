# Phase 2 — CI, publishing and adoption path: Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan
> task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Publish the kit's images and packages from `vust22/e2e-kit`, ship the four GitHub Actions
workflows spec §8 and §10 call for, and prove the reusable workflow runs end to end — without touching
the Mollie fork or the invertus org.

**Architecture:** Four workflows in the kit repo. `images.yml` builds the E2E CA and all three images
in one job and pushes them to GHCR as a matched tag set. `release.yml` runs changesets and publishes
the two packages to GitHub Packages under `@vust22/*` via a publish-time scope remap, leaving every
source import untouched. `e2e-reusable.yml` implements the §8.1 job graph by calling the same
`e2e-kit up/test/down` CLI a developer uses locally. `kit-ci.yml` gates kit PRs, and `e2e-selftest.yml`
calls the reusable workflow to prove it works.

**Tech Stack:** Node 22, TypeScript 5.9, zod 4, Playwright 1.62.1, Docker Compose, GitHub Actions,
changesets, GitHub Container Registry, GitHub Packages npm registry.

```ts
import type { E2EConfig } from './schema.js';

/** One `e2e-mock` job: a platform version and a Playwright shard. */
export interface CiMatrixEntry {
  ps: string;
  /** Playwright's `--shard` value, e.g. `1/2`. */
  shard: string;
  shardIndex: number;
}

export interface CiMatrix {
  moduleName: string;
  shards: number;
  mock: CiMatrixEntry[];
  sandbox: { ps: string }[];
  sandboxEnabled: boolean;
  sandboxBlocking: boolean;
  /**
   * Secret names the sandbox jobs need. Deliberately **empty when sandbox is off**: the reusable
   * workflow forwards exactly this list, and spec §11 requires mock jobs to run with zero provider
   * secrets present. Emitting names for a disabled sandbox would be the first step toward leaking
   * them into a mock job.
   */
  requiredSecrets: string[];
}

/**
 * Derive the CI job matrix from a validated config (spec §8.1).
 *
 * Pure by design — the reusable workflow's `prepare` job shells out to `e2e-kit ci-matrix`, and this
 * function is what that command prints. Keeping it free of I/O is what makes the matrix testable
 * without booting Docker.
 *
 * Sandbox expands over platform versions only, never over shards: a sandbox run talks to a real
 * provider, and sharding it multiplies external calls without improving coverage (spec §8.1).
 */
export function ciMatrix(config: E2EConfig): CiMatrix {
  const shards = config.ci.shards;
  const versions = config.platform.versions;

  const mock: CiMatrixEntry[] = [];
  for (const ps of versions) {
    for (let i = 1; i <= shards; i++) {
      mock.push({ ps, shard: `${i}/${shards}`, shardIndex: i });
    }
  }

  const sandboxEnabled = config.psp?.sandbox.enabled === true;

  return {
    moduleName: config.module.name,
    shards,
    mock,
    sandbox: sandboxEnabled ? versions.map((ps) => ({ ps })) : [],
    sandboxEnabled,
    sandboxBlocking: config.psp?.sandbox.blocking ?? false,
    requiredSecrets: sandboxEnabled ? config.psp!.sandbox.requiredSecrets : [],
  };
}
```

Create `scripts/prepare-publish.mjs`:

```js
#!/usr/bin/env node
/**
 * Stage a workspace package for publishing under a different npm scope
 * (DECISIONS.md D-022, design §4.2).
 *
 * GitHub Packages requires a package's scope to match the repository owner, so the kit cannot publish
 * `@invertus/*` from a `vust22`-owned repo. Renaming in the repo is not an option: 51 files import
 * `@invertus/e2e-core`, and they resolve through a workspace symlink keyed on that name.
 *
 * So the rename happens here, on a copy, at publish time. Two edits:
 *
 *   1. `name` → `@<scope>/<base>`
 *   2. every `@invertus/e2e-*` runtime dependency → an npm **alias**, keeping `@invertus/...` as the
 *      key and pointing it at `npm:@<scope>/...`
 *
 * Edit 2 is the load-bearing one. The compiled `dist` still contains `import '@invertus/e2e-core'`,
 * so the published package must declare a dependency that installs to *that* path. Without the alias
 * the published adapter resolves nothing at runtime.
 *
 * Deleting this script is the migration back to publishing `@invertus/*` directly.
 */

import { cpSync, mkdirSync, readFileSync, rmSync, writeFileSync } from 'node:fs';
import path from 'node:path';
import process from 'node:process';

const SOURCE_SCOPE = '@invertus/';

/** Dependency fields a consumer actually installs. `devDependencies` is deliberately excluded. */
const RUNTIME_DEP_FIELDS = ['dependencies', 'peerDependencies', 'optionalDependencies'];

/**
 * @param {Record<string, any>} manifest a parsed package.json
 * @param {string} scope target scope without `@` or `/`, e.g. `vust22`
 * @returns {Record<string, any>} a new manifest; the input is not mutated
 */
export function remapManifest(manifest, scope) {
  if (!manifest.name?.startsWith(SOURCE_SCOPE)) {
    throw new Error(
      `prepare-publish: expected a package named '${SOURCE_SCOPE}...', got '${manifest.name}'`,
    );
  }

  const out = structuredClone(manifest);
  out.name = `@${scope}/${manifest.name.slice(SOURCE_SCOPE.length)}`;

  for (const field of RUNTIME_DEP_FIELDS) {
    const deps = out[field];
    if (!deps) continue;
    for (const [name, range] of Object.entries(deps)) {
      if (!name.startsWith(SOURCE_SCOPE)) continue;
      const base = name.slice(SOURCE_SCOPE.length);
      // Key stays @invertus/* — see the note above.
      deps[name] = `npm:@${scope}/${base}@${range}`;
    }
  }

  return out;
}

function main() {
  const [packageDir, stagingDir] = process.argv.slice(2).filter((a) => !a.startsWith('--'));
  const scopeFlagIndex = process.argv.indexOf('--scope');
  const scope = scopeFlagIndex === -1 ? 'vust22' : process.argv[scopeFlagIndex + 1];

  if (!packageDir || !stagingDir) {
    console.error('usage: prepare-publish.mjs <packageDir> <stagingDir> [--scope <scope>]');
    process.exit(1);
  }

  const manifestPath = path.join(packageDir, 'package.json');
  const manifest = JSON.parse(readFileSync(manifestPath, 'utf8'));
  const remapped = remapManifest(manifest, scope);

  rmSync(stagingDir, { recursive: true, force: true });
  mkdirSync(stagingDir, { recursive: true });

  // Only the `files` allowlist is published, plus the manifest itself.
  for (const entry of manifest.files ?? []) {
    const from = path.join(packageDir, entry);
    cpSync(from, path.join(stagingDir, entry), { recursive: true });
  }
  writeFileSync(path.join(stagingDir, 'package.json'), `${JSON.stringify(remapped, null, 2)}\n`);

  console.log(`prepare-publish: ${manifest.name} -> ${remapped.name} staged in ${stagingDir}`);
}

if (import.meta.url === `file://${process.argv[1]}`) main();
```

# Publish the seeded platform images and the provider mock as a MATCHED SET.
#
# One job, deliberately — not a per-image matrix. DECISIONS.md D-009 generates the E2E CA per build,
# and the mock's `api.mollie.com` leaf certificate is signed by that same CA. Two jobs would produce
# two CAs, and TLS interception inside the compose network would fail with an unknown-issuer error
# that looks like anything but a build problem. The CA, all three images, and the push happen here or
# they are not consistent.
#

# single `image-set` value (see e2e-reusable.yml).

  push:
    branches: [main]
    paths:
      - 'images/**'
      - 'packages/prestashop/src/seed/**'
      - 'scripts/build-image.mjs'
      - 'scripts/gen-ca.mjs'
      - '.github/workflows/images.yml'

permissions:
  contents: read
  packages: write

concurrency:
  group: images-${{ github.ref }}
  cancel-in-progress: false

env:
  REGISTRY: ghcr.io
  OWNER: ${{ github.repository_owner }}
  E2E_PLATFORM: linux/amd64

jobs:
  publish:
    runs-on: ubuntu-latest
    timeout-minutes: 60
    outputs:
      image-set: ${{ steps.tags.outputs.suffix }}
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4

          node-version: 22
          cache: npm

      # has to compile before any image build copies the manifest in.
      - run: npx tsc --build

      - id: tags

        run: node scripts/build-image.mjs --all

      - name: Log in to GHCR

            local local_tag="$1" remote_base="$2" immutable="$3"

            docker push "$REGISTRY/$OWNER/$remote_base:$immutable"
            docker push "$REGISTRY/$OWNER/$remote_base:main"

- [ ] **Step 2: Write the publisher the workflow calls**

Create `scripts/publish-all.mjs`:

```js
#!/usr/bin/env node
/**
 * Publish every releasable package through the scope remap (DECISIONS.md D-022).
 *
 * `changesets/action` expects a single publish command, so this wraps the per-package staging:
 * for each package, stage a copy with the remapped manifest, then `npm publish` from the staging
 * directory. Publishing from the staging dir rather than the workspace is what keeps `@invertus/*`
 * in the repo and `@vust22/*` in the registry.
 */

import { spawnSync } from 'node:child_process';
import { readFileSync } from 'node:fs';
import path from 'node:path';
import process from 'node:process';

const PACKAGES = ['packages/core', 'packages/prestashop'];
const SCOPE = process.env.PUBLISH_SCOPE ?? 'vust22';
const REGISTRY = 'https://npm.pkg.github.com';

function runOrDie(cmd, args, opts = {}) {
  const r = spawnSync(cmd, args, { stdio: 'inherit', ...opts });
  if (r.status !== 0) {
    console.error(`FAILED: ${cmd} ${args.join(' ')}`);
    process.exit(r.status ?? 1);
  }
}

let published = false;
for (const pkg of PACKAGES) {
  const staging = path.join('/tmp', `publish-${path.basename(pkg)}`);
  runOrDie('node', ['scripts/prepare-publish.mjs', pkg, staging, '--scope', SCOPE]);

  const manifest = JSON.parse(readFileSync(path.join(staging, 'package.json'), 'utf8'));
  // Idempotent: a re-run after a partial failure must not abort on an already-published version.
  const view = spawnSync('npm', ['view', `${manifest.name}@${manifest.version}`, 'version', '--registry', REGISTRY], {
    encoding: 'utf8',
  });
  if (view.status === 0 && view.stdout.trim() === manifest.version) {
    console.log(`skip: ${manifest.name}@${manifest.version} is already published`);
    continue;
  }

  runOrDie('npm', ['publish', '--registry', REGISTRY, '--access', 'public'], { cwd: staging });
  published = true;
}

// changesets/action reads this to decide whether to run post-publish steps.
console.log(published ? 'published' : 'nothing to publish');
```

```yaml
# Merge every shard's blob report into one HTML report, bundle it with the summary and container
# logs, and upload it as a single zip (DECISIONS.md D-020, design §5.2).
#
# A zip rather than gh-pages because the reusable workflow runs under the *consumer's* GITHUB_TOKEN
# and has no write access anywhere else, and because artifacts behave the same on private repos and
# the Free plan — so nothing about the report path changes when this moves to a real org.
name: report-zip
description: Merge Playwright blob reports and upload a single report zip

inputs:
  mode:
    description: mock | sandbox — kept separate because the Playwright project name encodes the
      platform version but not the mode, so merging the two would be indistinguishable
    required: true
  blob-glob:
    description: Path glob of downloaded blob-report directories
    required: true
  artifact-name:
    description: Name of the uploaded artifact
    required: true

outputs:
  summary-markdown:
    description: Inline PR-comment table
    value: ${{ steps.summarise.outputs.markdown }}

runs:
  using: composite
  steps:
    - name: Merge blob reports
      shell: bash
      run: |
        set -euo pipefail
        shopt -s nullglob
        blobs=(${{ inputs.blob-glob }})
        if [ ${#blobs[@]} -eq 0 ]; then
          echo "no blob reports found for mode=${{ inputs.mode }}; nothing to merge"
          mkdir -p merged-report
          echo '{"projects":[]}' > merged-report/summary.json
          exit 0
        fi
        mkdir -p all-blobs
        cp -r "${blobs[@]}" all-blobs/
        npx playwright merge-reports --reporter html,json all-blobs > merged-report.json
        mkdir -p merged-report
        mv playwright-report/* merged-report/ 2>/dev/null || true
        mv merged-report.json merged-report/summary.json

    - id: summarise
      shell: bash
      run: |
        set -euo pipefail
        md=$(node -e '
          const fs = require("fs");
          let j; try { j = JSON.parse(fs.readFileSync("merged-report/summary.json","utf8")); }
          catch { console.log("_no report produced_"); process.exit(0); }
          const specs = (j.suites ?? []).flatMap(function walk(s) {
            return [...(s.specs ?? []), ...(s.suites ?? []).flatMap(walk)];
          });
          const rows = {};
          for (const spec of specs) for (const t of spec.tests ?? []) {
            const p = t.projectName ?? "unknown";
            rows[p] ??= { passed: 0, failed: 0, skipped: 0 };
            const s = t.status === "expected" ? "passed" : t.status === "skipped" ? "skipped" : "failed";
            rows[p][s]++;
          }
          const lines = ["| project | passed | failed | skipped |", "|---|---:|---:|---:|"];
          for (const [p, r] of Object.entries(rows))
            lines.push(`| \`${p}\` | ${r.passed} | ${r.failed} | ${r.skipped} |`);
          console.log(lines.join("\n"));
        ')
        {
          echo 'markdown<<REPORT_EOF'
          echo "$md"
          echo 'REPORT_EOF'
        } >> "$GITHUB_OUTPUT"
        echo "$md" >> "$GITHUB_STEP_SUMMARY"

    - uses: actions/upload-artifact@v4
      with:
        name: ${{ inputs.artifact-name }}
        path: |
          merged-report
          logs
        retention-days: 30
        if-no-files-found: warn
```
